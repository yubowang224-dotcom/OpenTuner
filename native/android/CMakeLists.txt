cmake_minimum_required(VERSION 3.22.1)
project(opentune_audio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(.. dsp)
add_subdirectory(../third_party/oboe oboe)
add_library(opentune_audio SHARED audio_engine.cpp)
target_include_directories(opentune_audio PRIVATE ../include)
target_link_libraries(opentune_audio PRIVATE opentune_dsp oboe android log)
target_compile_options(opentune_audio PRIVATE -Wall -Wextra -O2)
target_link_options(opentune_audio PRIVATE "-Wl,-z,max-page-size=16384")
