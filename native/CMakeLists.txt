cmake_minimum_required(VERSION 3.22.1)
project(opentune_dsp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(opentune_dsp STATIC src/analyzer.cpp src/piano_analyzer.cpp src/partial_analyzer.cpp src/c_api.cpp src/features.cpp src/pitch_tracker.cpp src/tone_synth.cpp src/metronome.cpp src/waveform.cpp)
set_target_properties(opentune_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(opentune_dsp PUBLIC include)
if(MSVC)
  target_compile_options(opentune_dsp PRIVATE /W4 /O2)
else()
  target_compile_options(opentune_dsp PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
if(NOT ANDROID)
  find_package(Threads REQUIRED)
  enable_testing()
  add_executable(dsp_tests tests/dsp_tests.cpp)
  target_link_libraries(dsp_tests PRIVATE opentune_dsp Threads::Threads)
  if(WIN32)
    target_link_libraries(dsp_tests PRIVATE psapi)
  endif()
  add_test(NAME production_dsp COMMAND dsp_tests ${CMAKE_BINARY_DIR}/results)
endif()
