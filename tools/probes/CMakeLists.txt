cmake_minimum_required(VERSION 3.22)
project(opentune_toolchain_probe LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
add_executable(native_probe native_probe.cpp)
target_compile_features(native_probe PRIVATE cxx_std_17)
target_link_libraries(native_probe PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(native_probe PRIVATE /W4 /WX)
else()
  target_compile_options(native_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME native_toolchain_smoke COMMAND native_probe)
